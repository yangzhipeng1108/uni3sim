cmake_minimum_required(VERSION 3.16)
project(mini_gemm_bench)
set(CMAKE_CXX_STANDARD 17)
set(ASCEND_HOME_PATH $ENV{ASCEND_HOME_PATH})
set(SOC_VERSION Ascend910B2)
set(ASCEND_CANN_PACKAGE_PATH ${ASCEND_HOME_PATH})
include(${ASCEND_HOME_PATH}/compiler/tikcpp/ascendc_kernel_cmake/ascendc.cmake)
foreach(SHAPE_NAME BM BN BK)
    if(NOT DEFINED ${SHAPE_NAME}_VAL OR NOT "${${SHAPE_NAME}_VAL}" MATCHES "^[1-9][0-9]*$")
        message(FATAL_ERROR "${SHAPE_NAME}_VAL must be a positive decimal integer")
    endif()
endforeach()
add_compile_definitions(BM=${BM_VAL} BN=${BN_VAL} BK=${BK_VAL})
ascendc_library(mini_gemm_kernel STATIC mini_gemm_kernel.cpp)
add_executable(mini_gemm_bench mini_gemm_main.cpp)
target_include_directories(mini_gemm_bench PRIVATE ${ASCEND_HOME_PATH}/include ${ASCEND_HOME_PATH}/runtime/include)
target_link_directories(mini_gemm_bench PRIVATE ${ASCEND_HOME_PATH}/lib64 ${ASCEND_HOME_PATH}/runtime/lib64)
target_link_libraries(mini_gemm_bench PRIVATE mini_gemm_kernel ascendcl)
