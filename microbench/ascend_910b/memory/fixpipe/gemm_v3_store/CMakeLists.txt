cmake_minimum_required(VERSION 3.16)
project(gemm_v3_store)
set(CMAKE_CXX_STANDARD 17)
if(NOT DEFINED ASCEND_HOME_PATH)
    set(ASCEND_HOME_PATH $ENV{ASCEND_HOME_PATH})
endif()
if(NOT DEFINED SOC_VERSION)
    set(SOC_VERSION "Ascend910B2")
endif()
if(NOT DEFINED ASCEND_CANN_PACKAGE_PATH)
    set(ASCEND_CANN_PACKAGE_PATH ${ASCEND_HOME_PATH})
endif()
include(${ASCEND_HOME_PATH}/compiler/tikcpp/ascendc_kernel_cmake/ascendc.cmake)
ascendc_library(gemm_v3_store_kernel STATIC gemm_v3_store_kernel.cpp)
add_executable(gemm_v3_store gemm_v3_store_main.cpp)
target_include_directories(gemm_v3_store PRIVATE
    ${ASCEND_HOME_PATH}/include ${ASCEND_HOME_PATH}/runtime/include)
target_link_directories(gemm_v3_store PRIVATE
    ${ASCEND_HOME_PATH}/lib64 ${ASCEND_HOME_PATH}/runtime/lib64)
target_link_libraries(gemm_v3_store PRIVATE gemm_v3_store_kernel
    ascendcl runtime mmpa c_sec stdc++ pthread dl)
